# SHTAPM edge runtime dependencies.
#
# This file covers what edge/main.py actually imports and executes (P1
# acquisition path) plus P2 anomaly-detection tooling. P3 (torch) is
# deliberately kept out — see edge/requirements-p3.txt.
#
# Not yet needed (no driver code imports them): gpiozero — two channels
# (pressure, gas) are still fake drivers (edge/drivers/fake.py); add this
# only when a real driver for one of them lands.
#
# humidity (edge/drivers/dht22.py) and temperature (edge/drivers/ds18b20.py)
# need no extra package: both read kernel-driver sysfs files (stdlib pathlib
# only) rather than a userspace library — see their module docstrings.
# w1thermsensor specifically is NOT used for the same reason (DS18B20 reads
# /sys/bus/w1/devices/28-*/w1_slave directly).

# P0/M3 (active): Shared contract serialization (frozen telemetry message).
# Edge imports app.schemas.contracts (backend Pydantic models); required to build
# TelemetryMessage on the edge for MQTT publication. Pinned to frozen stack.
pydantic==2.*

# P1 (active): MQTT telemetry/status publishing (edge/acquisition/mqtt_publisher.py).
# `ResilientTelemetryPublisher.start()` imports `paho.mqtt.client` — required for
# edge/main.py to actually run (import of the module succeeds without it; calling
# publisher.start() does not). Same pin as backend/requirements.txt.
paho-mqtt==1.6.*

# P1 (active): INA219 current sensor driver over I²C (edge/drivers/ina219.py).
smbus2==0.4.*

# P1 (active): ADXL335 vibration sensor driver over MCP3008/SPI (edge/drivers/adxl335.py).
spidev==3.8.*

# P2 (active): Isolation Forest detector (edge/anomaly/iforest.py, FR-A1).
# Upgraded to 1.5.* for Python 3.13 support (numpy 2.1.0+).
# Pulls in a compatible numpy transitively; no separate numpy pin here.
scikit-learn==1.5.*

# P2 (active): SWaT.A1 dataset harness (edge/eval/swat_eval.py, D011/D012).
# Reads the .xlsx dataset files; not a runtime dependency of the edge pipeline.
openpyxl==3.1.*
